/**
 * Login Function
 */
function checkLogin(username, password, role) {
  // CHANGED: Using openById to guarantee connection
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEET_NAMES.users);
  
  if (!sheet) return { status: 'error', message: 'User sheet not found' };
  
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (String(row[0]).toLowerCase() === String(username).toLowerCase() && 
        String(row[1]) === String(password)) {
      
      const userRole = String(row[2]).toLowerCase();
      const reqRole = String(role).toLowerCase();
      
      if (userRole === 'admin' || userRole === reqRole || (userRole === 'verifier' && reqRole === 'verifier')) {
         return { status: 'success', role: row[2] };
      }
    }
  }
  return { status: 'error', message: 'Invalid Credentials' };
}

/**
 * Fetch Data using Delta Logic (Extreme Performance)
 */
function getDashboardData(clientTimestamp) {
  const cache = CacheService.getScriptCache();
  let lastUpdate = cache.get('vgm_last_update');
  
  // 1. If no timestamp exists, create one
  if (!lastUpdate) {
    lastUpdate = Date.now().toString();
    cache.put('vgm_last_update', lastUpdate, 21600);
  }

  // 2. If client is already up-to-date, send NOTHING back! (Saves massive bandwidth)
  if (clientTimestamp && Number(clientTimestamp) >= Number(lastUpdate)) {
    return { changed: false };
  }

  // 3. Otherwise, fetch the fresh data from the sheet
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const getData = (sheetName) => {
    const sheet = ss.getSheetByName(sheetName);
    return sheet ? sheet.getDataRange().getValues().slice(1) : []; 
  };

  const freshData = {
    b17: getData(SHEET_NAMES.b17),
    b22: getData(SHEET_NAMES.b22),
    loma: getData(SHEET_NAMES.loma),
    b22seq: getData(SHEET_NAMES.b22seq),
    checkPart: getData(SHEET_NAMES.checkPart)
  };
  
  return { 
    changed: true, 
    data: freshData, 
    timestamp: lastUpdate 
  };
}

/**
 * Save Counts (Counter/Admin) with LockService for high traffic
 */
function saveData(datasetKey, partNo, values) {
  const lock = LockService.getScriptLock();
  try {
    // Wait up to 15 seconds for other users to finish saving
    lock.waitLock(15000); 

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetName = SHEET_NAMES[datasetKey];
    const conf = SHEET_MAP[datasetKey];
    const sheet = ss.getSheetByName(sheetName);

    const data = sheet.getDataRange().getValues();
    let rowIndex = -1;
    for(let i = 0; i < data.length; i++) {
      if(String(data[i][conf.partNo]).trim() === String(partNo).trim()) {
        rowIndex = i + 1; 
        break;
      }
    }
    
    if (rowIndex === -1) return { status: 'error', message: 'Part not found' };

    // Write Boxes and Recounts
    sheet.getRange(rowIndex, conf.boxStart + 1, 1, 5).setValues([[values.box1, values.box2, values.box3, values.box4, values.box5]]);
    sheet.getRange(rowIndex, conf.recStart + 1, 1, 5).setValues([[values.recount1, values.recount2, values.recount3, values.recount4, values.recount5]]);

    const currentStatus = sheet.getRange(rowIndex, conf.status + 1).getValue();
    if (String(currentStatus).toLowerCase() !== 'verified' && (values.box1 !== "" || values.box2 !== "")) {
      sheet.getRange(rowIndex, conf.status + 1).setValue("Counted");
    }

    // Tell all other dashboards that data has changed
    CacheService.getScriptCache().put('vgm_last_update', Date.now().toString(), 21600);
    return { status: 'success' };

  } catch (e) {
    return { status: 'error', message: 'Database busy (Timeout). Please try saving again.' };
  } finally {
    // Always release the lock so the next person can save
    lock.releaseLock();
  }
}

/**
 * Verify Item with LockService
 */
function verifyItem(datasetKey, partNo, verifierName) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const conf = SHEET_MAP[datasetKey];
    const sheet = ss.getSheetByName(SHEET_NAMES[datasetKey]);
    const data = sheet.getDataRange().getValues();
    
    for(let i=0; i<data.length; i++) {
       if(String(data[i][conf.partNo]).trim() === String(partNo).trim()) {
         sheet.getRange(i+1, conf.status + 1).setValue("Verified");
         sheet.getRange(i+1, conf.verified + 1).setValue(verifierName);
         // Tell all other dashboards that data has changed
         CacheService.getScriptCache().put('vgm_last_update', Date.now().toString(), 21600);
         return { status: 'success' };
       }
    }
    return { status: 'error', message: 'Part not found' };
  } catch (e) {
    return { status: 'error', message: 'Server busy. Try again.' };
  } finally {
    lock.releaseLock();
  }
}
