      function changeLanguage(lang) {

      function handleGesture() {
          const xDiff = touchEndX - touchStartX;
          const yDiff = touchEndY - touchStartY;
          if (yDiff > 120 && window.scrollY <= 0 && Math.abs(xDiff) < 60) {
              handleRefresh();
          }
      }

      function updateCardStats(key, dataset) {
        const total = dataset.length;
        const txtEl = document.getElementById(`stat-txt-${key}`);
        const pctEl = document.getElementById(`stat-pct-${key}`);
        const barEl = document.getElementById(`prog-bar-${key}`);
        const badgeEl = document.getElementById(`stat-badge-${key}`); // <-- Grabs the new badge ID
        
        const conf = SHEET_CONFIG[key];

        // IF NO DATA EXISTS YET (Or sheet is empty)
        if (total === 0) {
          if(txtEl) txtEl.innerText = "No Data";
          if(pctEl) pctEl.innerText = "0%";
          if(barEl) barEl.style.width = "0%";
          if(badgeEl) {
              badgeEl.innerText = "Pending";
              badgeEl.className = "status-badge status-not-counted"; // Turns the badge red
          }
          return;
        }

        let progressCount = 0;
        dataset.forEach(r => {
          const status = String(r[conf.status] || "").trim().toLowerCase();
          if (status === 'verified' || status === 'counted') {
            progressCount++;
          }
        });

        // IF DATA IS SUCCESSFULLY LOADED
        const percentage = Math.round((progressCount / total) * 100);
        if(txtEl) txtEl.innerText = `${progressCount} / ${total} Items`;
        if(pctEl) pctEl.innerText = `${percentage}%`;
        if(barEl) barEl.style.width = `${percentage}%`;
        if(badgeEl) {
            badgeEl.innerText = "Ready";
            badgeEl.className = "status-badge status-counted"; // Turns the badge blue
        }
      }

      function startLiveUpdate() {
        document.getElementById('live-indicator').style.display = 'flex';
        if (pollingInterval) clearInterval(pollingInterval);
        
        // Use a random offset between 0 and 3 seconds so users are staggered
        const jitter = Math.floor(Math.random() * 3000);
        
        pollingInterval = setInterval(() => {
          if (isFetching) return;
          if (document.getElementById('single-item-view').style.display === 'block') return; 

          isFetching = true;
          google.script.run.withSuccessHandler(data => {
            globalData = data; 
            isFetching = false;
            if(document.getElementById('selection-view').style.display === 'block') renderDashboard(globalData);
            if (document.getElementById('detail-view').style.display === 'block') renderCurrentRackList();
          }).withFailureHandler(() => { isFetching = false; }).getDashboardData();
        }, 5000 + jitter); // 5 seconds + a random small delay
      }

      function openUserProgress() {
          document.getElementById('selection-view').style.display = 'none';
          document.getElementById('user-progress-view').style.display = 'block';
          document.getElementById('app-container').classList.add('wide');
          
          document.getElementById('up-name').innerText = currentUsername;
          document.getElementById('up-role').innerText = currentUserRole;
          document.getElementById('up-login').innerText = loginTimestamp || "Current Session";
          document.getElementById('up-logout').innerText = localStorage.getItem('lastLogout') || "-";
          
          let totalProcessed = 0;
          let itemsList = [];
          
          const keys = ['b17', 'b22', 'loma', 'b22seq'];
          
          const roleLower = currentUserRole.toLowerCase();
          const isVerifier = roleLower.includes('verifier');
          const targetUsername = currentUsername.toLowerCase().trim();
          
          document.getElementById('up-stat-label').innerText = isVerifier ? "Items Verified by You" : "Items Counted";

          keys.forEach(key => {
             const data = globalData[key] || [];
             const conf = SHEET_CONFIG[key];
             
             data.forEach(r => {
                 let match = false;
                 
                 const rowStatus = String(r[conf.status] || "").trim();
                 const rowVerifiedBy = String(r[conf.verified] || "").trim().toLowerCase();

                 if (isVerifier) {
                     // STRICT VERIFIER RULE:
                     // 1. Status must be "Verified".
                     // 2. "Verified By" column must match current username.
                     if (rowStatus === 'Verified' && rowVerifiedBy === targetUsername) {
                         match = true;
                     }
                 } else {
                     // STRICT COUNTER RULE:
                     // 1. Status must be "Counted".
                     // 2. Counters NEVER see "Verified" items.
                     // Since there is no "Counted By" column in your sheet, counters see all currently "Counted" items.
                     if (rowStatus === 'Counted') {
                         match = true;
                     }
                 }

                 if (match) {
                     totalProcessed++;
                     itemsList.push({
                         material: r[conf.material],
                         name: r[conf.name],
                         rack: r[conf.rack] || r[conf.tact] || "-", 
                         status: rowStatus,
                         key: key
                     });
                 }
             });
          });
          
          document.getElementById('up-stat-count').innerText = totalProcessed;
          
          // Populate Table
          const tbody = document.getElementById('user-detail-tbody');
          tbody.innerHTML = "";
          if (itemsList.length === 0) {
              tbody.innerHTML = "<tr><td colspan='4' style='text-align:center'>No activity matches your role rules.</td></tr>";
          } else {
              itemsList.forEach(item => {
                  let badge = "status-counted";
                  if(item.status === 'Verified') badge = "status-verified";
                  
                  tbody.innerHTML += `<tr>
                      <td>${item.material}</td>
                      <td>${item.name}</td>
                      <td>${item.rack}</td>
                      <td><span class="status-badge ${badge}">${item.status}</span></td>
                  </tr>`;
              });
          }
      }

             }).saveData(currentDatasetKey, currentItemPartNo, newValues);

      function openRackList(datasetKey) {
        currentDatasetKey = datasetKey;
        if (!globalData[datasetKey]) { alert("Data belum siap."); return; }
        
        document.getElementById('selection-view').style.display = 'none';
        document.getElementById('detail-view').style.display = 'block';
        document.getElementById('app-container').classList.add('wide');
        document.getElementById('search-input').value = "";
        
        const data = globalData[datasetKey];
        const conf = SHEET_CONFIG[datasetKey];
        
        const uniqueFilters = new Set();
        data.forEach(r => { 
            let groupIdx = (conf.rack !== null) ? conf.rack : conf.tact;
            let val = (r[groupIdx] || "").toString().trim(); 
            if(val) uniqueFilters.add(val);
        });
        const sortedFilters = Array.from(uniqueFilters).sort((a,b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' }));
        const select = document.getElementById('filter-select');
        select.innerHTML = `<option value="All">All Groups</option>`;
        sortedFilters.forEach(f => { select.innerHTML += `<option value="${f}">${f}</option>`; });
        
        renderCurrentRackList();
      }

      function renderCurrentRackList() {
        const data = globalData[currentDatasetKey];
        const conf = SHEET_CONFIG[currentDatasetKey];
        
        const container = document.getElementById('rack-container');
        const filterVal = document.getElementById('filter-select').value;
        const searchVal = document.getElementById('search-input').value.toLowerCase();
        
        container.innerHTML = "";
        
        let groupIdx = (conf.rack !== null) ? conf.rack : conf.tact;
        let groupLabel = (conf.rack !== null) ? t('lbl_rack') : "Tact";

        let filteredData = data.filter(r => {
            const group = (r[groupIdx] || "").toString().trim();
            const material = (r[conf.material] || "").toString().toLowerCase();
            const partNo = (r[conf.partNo] || "").toString().toLowerCase();
            const matchesFilter = (filterVal === 'All') || (group === filterVal);
            const matchesSearch = (searchVal === "") || material.includes(searchVal) || partNo.includes(searchVal);
            return matchesFilter && matchesSearch;
        });

        let countProgress = 0;
        let totalItems = filteredData.length;
        filteredData.forEach(r => {
             const s = String(r[conf.status] || "").trim().toLowerCase();
             if(s==='verified' || s==='counted') countProgress++;
        });
        const pct = totalItems === 0 ? 0 : Math.round((countProgress/totalItems)*100);
        document.getElementById('detail-prog-bar').style.width = pct + "%";
        document.getElementById('detail-prog-pct').innerText = pct + "% (" + countProgress + "/" + totalItems + ")";

        if(filteredData.length === 0) {
            container.innerHTML = `<div style="text-align:center; padding:20px;">${t('no_data')}</div>`;
            return;
        }

        // Changed class to location-section
        let html = `<div class="location-section"><table class="table"><thead><tr>
            <th data-lang="lbl_partno">${t('lbl_partno')}</th>
            <th data-lang="lbl_material">${t('lbl_material')}</th>
            <th>${groupLabel}</th>
            <th data-lang="lbl_status">${t('lbl_status')}</th>
        </tr></thead><tbody>`;

        filteredData.forEach(r => {
            const status = r[conf.status] || "Not Counted";
            let badgeClass = "status-not-counted";
            if(status === "Verified") badgeClass = "status-verified";
            else if(status === "Counted") badgeClass = "status-counted";

            const rowJson = encodeURIComponent(JSON.stringify(r));
            html += `<tr class="clickable-row" onclick="openSingleItemView('${rowJson}')">
                <td><b>${r[conf.partNo]}</b></td><td>${r[conf.material]}</td><td>${r[groupIdx]}</td>
                <td><span class="status-badge ${badgeClass}">${status}</span></td>
            </tr>`;
        });
        html += `</tbody></table></div>`;
        container.innerHTML = html;
      }

      function openProgressView() {
          document.getElementById('detail-view').style.display = 'none';
          document.getElementById('progress-view').style.display = 'block';
          
          if(myChart) myChart.destroy();
          
          const ctx = document.getElementById('progressChart').getContext('2d');
          const data = globalData[currentDatasetKey];
          const conf = SHEET_CONFIG[currentDatasetKey];
          
          if (currentDatasetKey === 'b17') {
              const racks = {}; 
              data.forEach(r => {
                  const rack = r[conf.rack] || "Unknown"; 
                  const status = String(r[conf.status]).trim().toLowerCase();
                  if(!racks[rack]) racks[rack] = { v:0, c:0, n:0 };
                  if(status === 'verified') racks[rack].v++;
                  else if(status === 'counted') racks[rack].c++;
                  else racks[rack].n++;
              });
              
              const labels = Object.keys(racks).sort(); 
              const dataV = labels.map(l => racks[l].v);
              const dataC = labels.map(l => racks[l].c);
              const dataN = labels.map(l => racks[l].n);
              
              myChart = new Chart(ctx, {
                  type: 'bar',
                  data: {
                      labels: labels,
                      datasets: [
                          { label: 'Not Counted', data: dataN, backgroundColor: '#e74c3c' },
                          { label: 'Counted', data: dataC, backgroundColor: '#3498db' },
                          { label: 'Verified', data: dataV, backgroundColor: '#2ecc71' }
                      ]
                  },
                  options: {
                      responsive: true,
                      maintainAspectRatio: false,
                      scales: {
                          x: { stacked: true },
                          y: { stacked: true, beginAtZero: true }
                      },
                      plugins: {
                          // Changed Rack to Location
                          title: { display: true, text: 'Status by Location (B17)' },
                          legend: { position: 'top' }
                      }
                  }
              });

          } else {
              let v=0, c=0, n=0;
              data.forEach(r => {
                  const status = String(r[conf.status]).trim().toLowerCase();
                  if(status === 'verified') v++;
                  else if(status === 'counted') c++;
                  else n++;
              });
              
              myChart = new Chart(ctx, {
                  type: 'bar',
                  data: {
                      labels: ['Verified', 'Counted', 'Not Counted'],
                      datasets: [{
                          label: 'Items',
                          data: [v, c, n],
                          backgroundColor: ['#2ecc71', '#3498db', '#e74c3c']
                      }]
                  },
                  options: {
                      responsive: true,
                      maintainAspectRatio: false,
                      scales: { y: { beginAtZero: true } },
                      plugins: { legend: { display: false } }
                  }
              });
          }
      }

      function openSingleItemView(rowEncoded) {

      function validateAndLock(input) {
         const val = input.value;
         const errDiv = input.closest('.detail-value').querySelector('.error-msg'); 
         
         if (val.trim() === "") return; 
         
         // STRICT RULE 1: Recount != Box
         if(input.id.includes('val-rec')) {
             const idx = input.id.replace('val-rec', '');
             const boxInput = document.getElementById('val-box'+idx);
             // Ensure box exists and has value (including "0") before comparing
             if(boxInput && boxInput.value !== "" && Number(val) === Number(boxInput.value)) {
                 input.classList.add('input-error');
                 if(errDiv) {
                     errDiv.innerText = "Recount cannot be same as Box value.";
                     errDiv.style.display = 'block';
                 }
                 input.value = ""; // Clear
                 return;
             }
         }

         const num = Number(val);
         if (!/^\d+$/.test(val) || num < 0) {
             showInlineError(input, errDiv, "Only integers ≥ 0 allowed.");
             input.value = ""; 
             return;
         }

         input.disabled = true;
         updateWaterfall();
      }

      function showInlineError(input, errDiv, msg) {
         input.classList.add('input-error');
         if(errDiv) {
             errDiv.innerText = msg;
             errDiv.style.display = 'block';
         }
         setTimeout(() => {
             clearError(input);
         }, 3000);
      }

      function clearError(input) {
         const errDiv = input.closest('.detail-value').querySelector('.error-msg'); 
         input.classList.remove('input-error');
         if(errDiv) errDiv.style.display = 'none';
      }

      function saveItem() {
         const btn = document.querySelector('.btn-save');
         btn.innerText = t('btn_processing'); btn.disabled = true;
         const newValues = getFormValues();
         const conf = SHEET_CONFIG[currentDatasetKey];

         google.script.run.withSuccessHandler(res => {
                if (res.status === 'success') {
                    alert(t('alert_saved'));
                    updateLocalData(currentItemPartNo, newValues);
                    const row = globalData[currentDatasetKey].find(x => x[conf.partNo] == currentItemPartNo);
                    if(row) openSingleItemView(encodeURIComponent(JSON.stringify(row)));
                } else {
                    alert("Error: " + res.message);
                    btn.innerText = t('btn_save'); btn.disabled = false;
                }
            }).saveData(currentDatasetKey, currentItemPartNo, newValues);
      }

      function verifyAndSaveVerifier() {
          if(!confirm("Verify and Save this item?")) return;
          const newValues = getFormValues(); 
          const conf = SHEET_CONFIG[currentDatasetKey];

          google.script.run.withSuccessHandler(res => {
              if(res.status === 'success') {
                  updateLocalData(currentItemPartNo, newValues);
                  google.script.run.withSuccessHandler(vRes => {
                      if(vRes.status === 'success') {
                          alert(t('alert_verified'));
                          const row = globalData[currentDatasetKey].find(x => x[conf.partNo] == currentItemPartNo);
                          if(row) {
                              row[conf.status] = "Verified"; row[conf.verified] = currentUsername;
                              openSingleItemView(encodeURIComponent(JSON.stringify(row)));
                          }
                      }
                  }).verifyItem(currentDatasetKey, currentItemPartNo, currentUsername);
              } else {
                  alert("Error Saving Recounts: " + res.message);
              }
          }).saveData(currentDatasetKey, currentItemPartNo, newValues);
      }

      function verifyItem() {
         if(!confirm("Are you sure you want to verify this item?")) return;
         const conf = SHEET_CONFIG[currentDatasetKey];
         
         google.script.run.withSuccessHandler(res => {
             if(res.status === 'success') {
                 alert(t('alert_verified'));
                 const row = globalData[currentDatasetKey].find(x => x[conf.partNo] == currentItemPartNo);
                 if(row) {
                    row[conf.status] = "Verified"; row[conf.verified] = currentUsername;

                 }
             } else { alert("Error: " + res.message); }

      function updateLocalData(partNo, values) {
          const conf = SHEET_CONFIG[currentDatasetKey];
          const row = globalData[currentDatasetKey].find(x => x[conf.partNo] == partNo);
          if (row) {
            conf.boxes.forEach((idx, i) => { row[idx] = values[`box${i+1}`]; });
            conf.recounts.forEach((idx, i) => { row[idx] = values[`recount${i+1}`]; });
            if (String(row[conf.status]) !== "Verified" && values.box1 !== "") { 
                row[conf.status] = "Counted"; 
            }
          }
      }
